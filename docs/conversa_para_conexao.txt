Resposta do agente
------------------------------------------------------------------------

Boa, você está no caminho certo. O segredo é entender a ordem das
camadas e que cada uma só enxerga a próxima:

  navegador (fetch)
     -> src/routes/comentarios.js              (define a URL)
     -> controllers/comentarioController.js     (lê req, valida, responde res)
     -> models/comentarioModel.js               (monta e executa o SQL)
     -> database/config.js                      (conexão MySQL)
     -> tabelas comentario / locais / usuario

Depois de criar o comentarioController.js, você precisa fazer 4 coisas,
nesta ordem. Cada uma é explicada abaixo, com código no padrão do
projeto (para você mesmo digitar).

PASSO 1 — TERMINAR O comentarioModel.js (FAZER PRIMEIRO)

Atenção a um bug que vai quebrar o servidor: na linha 24 você exporta
cadastrar, mas a função cadastrar NÃO existe no arquivo. Assim que um
controller der require("../models/comentarioModel"), o Node lança
ReferenceError: cadastrar is not defined.

O model correto deve ter 3 funções:

var database = require("../database/config");

function listar() {
    var instrucaoSql = `
        SELECT c.id, c.nota, c.comentario, c.fk_local, c.fk_usuario,
               l.nome AS nome_local, u.nome AS nome_usuario
        FROM comentario c
        JOIN locais l  ON l.id = c.fk_local
        JOIN usuario u ON u.id = c.fk_usuario;
    `;
    console.log("Executando a instrução SQL: \n" + instrucaoSql);
    return database.executar(instrucaoSql);
}

function listar_por_local(id_local) {
    var instrucaoSql = `
        SELECT c.id, c.nota, c.comentario, c.fk_usuario, u.nome AS nome_usuario
        FROM comentario c
        JOIN usuario u ON u.id = c.fk_usuario
        WHERE c.fk_local = ${id_local};
    `;
    console.log("Executando a instrução SQL: \n" + instrucaoSql);
    return database.executar(instrucaoSql);
}

function cadastrar(nota, comentario, fk_local, fk_usuario) {
    var instrucaoSql = `
        INSERT INTO comentario (nota, comentario, fk_local, fk_usuario)
        VALUES ('${nota}', '${comentario}', ${fk_local}, ${fk_usuario});
    `;
    console.log("Executando a instrução SQL: \n" + instrucaoSql);
    return database.executar(instrucaoSql);
}

module.exports = {
    listar,
    listar_por_local,
    cadastrar
};

Conceito importante: no seu banco_so_site.sql, nota é CHAR(5) (texto)
-> vai entre aspas. fk_local e fk_usuario são INT -> sem aspas. As
funções antigas buscarPorId e buscarPorCnpj do template não servem para
o site; pode removê-las (mas confirme antes, porque é remoção de
código).

PASSO 2 — REESCREVER O comentarioController.js

O arquivo atual ainda é do template: faz
require("../models/empresaModel"), e esse arquivo foi deletado (aparece
como D no git status). Se uma rota carregar esse controller, dá
MODULE_NOT_FOUND. Ele precisa apontar para o comentarioModel que você
acabou de arrumar:

var comentarioModel = require("../models/comentarioModel");

function listar(req, res) {
    comentarioModel.listar().then(function (resultado) {
        res.status(200).json(resultado);
    }).catch(function (erro) {
        console.log(erro);
        console.log("Houve um erro ao buscar os comentários: ", erro.sqlMessage);
        res.status(500).json(erro.sqlMessage);
    });
}

function listar_por_local(req, res) {
    var id_local = req.params.id_local;

    comentarioModel.listar_por_local(id_local).then(function (resultado) {
        res.status(200).json(resultado);
    }).catch(function (erro) {
        console.log(erro);
        console.log("Houve um erro ao buscar os comentários: ", erro.sqlMessage);
        res.status(500).json(erro.sqlMessage);
    });
}

function cadastrar(req, res) {
    var nota = req.body.nota;
    var comentario = req.body.comentario;
    var fk_local = req.body.fk_local;
    var fk_usuario = req.body.fk_usuario;

    if (nota == undefined) {
        res.status(400).send("A nota está undefined!");
    } else if (comentario == undefined) {
        res.status(400).send("O comentário está undefined!");
    } else if (fk_local == undefined) {
        res.status(400).send("O local está undefined!");
    } else if (fk_usuario == undefined) {
        res.status(400).send("O usuário está undefined!");
    } else {
        comentarioModel.cadastrar(nota, comentario, fk_local, fk_usuario)
            .then(function (resultado) {
                res.status(201).json(resultado);
            }).catch(function (erro) {
                console.log(erro);
                console.log("Houve um erro ao cadastrar o comentário: ", erro.sqlMessage);
                res.status(500).json(erro.sqlMessage);
            });
    }
}

module.exports = {
    listar,
    listar_por_local,
    cadastrar
};

Conceitos:
  - req.params.id_local -> vem NA URL (/comentarios/por_local/1).
  - req.body.nota -> vem no CORPO da requisição (JSON do fetch/Postman).
  - .then() = sucesso, .catch() = erro do banco.
  - Status: 200 ok, 201 criado, 400 dados inválidos, 500 erro no
    servidor/MySQL.

PASSO 3 — CRIAR A ROTA src/routes/comentarios.js

A rota só liga a URL ao controller. Crie o arquivo seguindo o padrão de
src/routes/avisos.js:

var express = require("express");
var router = express.Router();

var comentarioController = require("../controllers/comentarioController");

router.get("/listar", function (req, res) {
    comentarioController.listar(req, res);
});

router.get("/por_local/:id_local", function (req, res) {
    comentarioController.listar_por_local(req, res);
});

router.post("/cadastrar", function (req, res) {
    comentarioController.cadastrar(req, res);
});

module.exports = router;

PASSO 4 — MONTAR A ROTA NO app.js

O app.js só conhece as rotas que você registra com app.use(...). Existe
uma função tryMount para isso. Logo depois da linha que monta /locais
(linha 47), adicione:

    tryMount('./src/routes/comentarios', '/comentarios');

Sem isso, a URL /comentarios/... cai no handler de 404 (Not Found).

ATENÇÃO: o tryMount tem try/catch que ignora erros silenciosamente. Se
o seu controller/model tiver erro de sintaxe, você não verá o erro e a
rota simplesmente não sobe. Se /comentarios/listar retornar Not Found, o
problema está no require de algum arquivo da cadeia.

PASSO 5 — CORRIGIR O BACKEND DE USUÁRIO (SENÃO LOGIN/CADASTRO NÃO
FUNCIONAM)

Comentário precisa de fk_usuario, então o usuário tem que existir/logar.
Hoje há 3 problemas:

1. src/models/usuarioModel.js — o SELECT do autenticar busca fk_empresa,
   coluna que não existe no seu banco. Troque para:

       SELECT id, nome, email FROM usuario
       WHERE email = '${email}' AND senha = '${senha}';

2. src/controllers/usuarioController.js — no autenticar, a linha que
   chama buscarAquariosPorEmpresa(...) usa uma função que não foi
   importada (o git diff mostra que você trocou
   aquarioModel.buscarAquariosPorEmpresa por buscarAquariosPorEmpresa)
   -> isso gera ReferenceError no login. O login deve apenas responder
   id, nome, email, sem "locais/aquarios".

3. src/controllers/usuarioController.js — no cadastrar, há a validação
   fkEmpresa == undefined que devolve 400. O formulário do site
   (ipt_nome, ipt_email, ipt_senha) não manda empresa. Remova essa
   validação e a variável fkEmpresa.

PASSO 6 — CONSERTAR src/routes/locais.js (OPCIONAL PARA COMENTÁRIOS)

Esse arquivo está com erro de sintaxe:
  - require("SELECT * from locais") -> require espera caminho de
    arquivo, não SQL.
  - usa aquarioController, que não foi importado.

Por isso a rota /locais nunca sobe (o tryMount engole o erro). Você não
precisa dela para os comentários, porque os dashboards já usam os ids
fixos 1, 2 e 3. Duas opções: criar um locaisController de verdade, ou
remover esse arquivo e a linha
tryMount('./src/routes/locais', '/locais') do app.js. Me diga qual
prefere antes de mexer.

PASSO 7 — TESTAR (NESTA ORDEM)

    npm start

O servidor sobe em http://localhost:3333 (porta do .env.dev). Depois, em
outro terminal:

    # 1) cadastrar um usuário
    curl -X POST http://localhost:3333/usuarios/cadastrar \
      -H "Content-Type: application/json" \
      -d "{\"nomeServer\":\"Pedro\",\"emailServer\":\"pedro@email.com\",\"senhaServer\":\"Senha123\"}"

    # 2) cadastrar um comentário (use um id de usuário que exista e local 1)
    curl -X POST http://localhost:3333/comentarios/cadastrar \
      -H "Content-Type: application/json" \
      -d "{\"nota\":\"5\",\"comentario\":\"Otimo lugar\",\"fk_local\":1,\"fk_usuario\":1}"

    # 3) listar comentários do local 1
    curl http://localhost:3333/comentarios/por_local/1

Se o POST retornar 201 e o GET retornar o JSON, o backend está
conversando com o banco. Só depois disso faz sentido ligar o fetch nos
HTMLs do site/.

  rota comentarios.js -> montar no app.js -> corrigir usuário -> testar
